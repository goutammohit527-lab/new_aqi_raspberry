// ========================================================= // GRAPH DATA // ========================================================= let aqiHistory = []; const MAX_GRAPH_POINTS = 40; // ========================================================= // ELEMENTS // ========================================================= const statusDot = document.getElementById("statusDot"); const statusText = document.getElementById("statusText"); const clockElement = document.getElementById("clock"); const aqiValue = document.getElementById("aqiValue"); const aqiStatus = document.getElementById("aqiStatus"); const graphLine = document.getElementById("graphLine"); const lastUpdate = document.getElementById("lastUpdate"); // ========================================================= // CLOCK // ========================================================= function updateClock() { const now = new Date(); const hours = String(now.getHours()).padStart(2, "0"); const minutes = String(now.getMinutes()).padStart(2, "0"); const seconds = String(now.getSeconds()).padStart(2, "0"); clockElement.textContent = `${hours}:${minutes}:${seconds}`; } setInterval(updateClock, 1000); updateClock(); // ========================================================= // SAFE VALUE // ========================================================= function displayValue(value, decimals = 0) { if ( value === null || value === undefined || value === "" || Number.isNaN(Number(value)) ) { return "--"; } return Number(value).toFixed(decimals); } // ========================================================= // UPDATE SENSOR VALUE // ========================================================= function setValue(id, value, decimals = 0) { const element = document.getElementById(id); if (!element) { return; } element.textContent = displayValue(value, decimals); } // ========================================================= // CONNECTION STATUS // ========================================================= function setConnected() { statusDot.classList.remove("disconnected"); statusDot.classList.add("connected"); statusText.textContent = "Connected"; } function setDisconnected() { statusDot.classList.remove("connected"); statusDot.classList.add("disconnected"); statusText.textContent = "Disconnected"; } // ========================================================= // AQI STATUS TEXT // ========================================================= function getAQIStatus(aqi) { if ( aqi === null || aqi === undefined || Number.isNaN(Number(aqi)) ) { return "Waiting for sensor..."; } const value = Number(aqi); if (value <= 50) { return "Good"; } if (value <= 100) { return "Satisfactory"; } if (value <= 200) { return "Moderate"; } if (value <= 300) { return "Poor"; } if (value <= 400) { return "Very Poor"; } return "Severe"; } // ========================================================= // GRAPH // ========================================================= function updateGraph(value) { if ( value === null || value === undefined || Number.isNaN(Number(value)) ) { return; } aqiHistory.push(Number(value)); if (aqiHistory.length > MAX_GRAPH_POINTS) { aqiHistory.shift(); } if (aqiHistory.length < 2) { return; } const width = 800; const height = 220; const padding = 10; const minValue = Math.min(...aqiHistory); const maxValue = Math.max(...aqiHistory); let range = maxValue - minValue; if (range === 0) { range = 1; } const points = aqiHistory.map((item, index) => { const x = padding + ( index / (aqiHistory.length - 1) ) * (width - padding * 2); const normalized = (item - minValue) / range; const y = height - padding - normalized * (height - padding * 2); return `${x},${y}`; }); graphLine.setAttribute( "points", points.join(" ") ); } // ========================================================= // LAST UPDATE // ========================================================= function updateLastUpdate() { const now = new Date(); const hours = String(now.getHours()).padStart(2, "0"); const minutes = String(now.getMinutes()).padStart(2, "0"); const seconds = String(now.getSeconds()).padStart(2, "0"); lastUpdate.textContent = `${hours}:${minutes}:${seconds}`; } // ========================================================= // UPDATE DASHBOARD // ========================================================= function updateDashboard(data) { if (!data) { return; } // ----------------------------- // AQI // ----------------------------- if ( data.aqi !== null && data.aqi !== undefined ) { aqiValue.textContent = displayValue(data.aqi, 0); aqiStatus.textContent = getAQIStatus(data.aqi); updateGraph(data.aqi); } else { aqiValue.textContent = "--"; aqiStatus.textContent = "Waiting for sensor..."; } // ----------------------------- // PM VALUES // ----------------------------- setValue( "pm1_0", data.pm1_0, 1 ); setValue( "pm2_5", data.pm2_5, 1 ); setValue( "pm10", data.pm10, 1 ); // ----------------------------- // GAS VALUES // ----------------------------- setValue( "co2", data.co2, 0 ); setValue( "tvoc", data.tvoc, 0 ); setValue( "co", data.co, 2 ); setValue( "ch2o", data.ch2o, 3 ); setValue( "o3", data.o3, 3 ); setValue( "no2", data.no2, 3 ); // ----------------------------- // TEMPERATURE // ----------------------------- setValue( "temperature", data.temperature, 1 ); // ----------------------------- // HUMIDITY // ----------------------------- setValue( "humidity", data.humidity, 1 ); // ----------------------------- // STATUS // ----------------------------- setConnected(); updateLastUpdate(); } // ========================================================= // FETCH SENSOR DATA // ========================================================= async function fetchSensorData() { try { const response = await fetch( API_URL, { method: "GET", cache: "no-store" } ); if (!response.ok) { throw new Error( `HTTP ${response.status}` ); } const data = await response.json(); updateDashboard(data); } catch (error) { console.error( "Sensor connection error:", error ); setDisconnected(); /* * Do not generate fake values. * Keep the previous graph/data state. */ aqiValue.textContent = "--"; aqiStatus.textContent = "Sensor unavailable"; setValue("pm1_0", null); setValue("pm2_5", null); setValue("pm10", null); setValue("co2", null); setValue("tvoc", null); setValue("co", null); setValue("ch2o", null); setValue("o3", null); setValue("no2", null); setValue("temperature", null); setValue("humidity", null); } } // ========================================================= // START // ========================================================= fetchSensorData(); // Read new sensor data every 2 seconds setInterval( fetchSensorData, 2000 ); ``` **The main visual change is in `style.css`:** each sensor card now has the sensor name at the top and the **actual reading + unit centered in the middle**, with the reading at `32px` and the unit at `14px`. After replacing the files, restart the display: ```bash sudo systemctl restart aqi-display.service ``` If the browser is already open in kiosk mode, the updated page should appear after Chromium reloads.
